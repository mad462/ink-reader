# EPUB MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the placeholder EPUB reader flow with a minimal usable EPUB open-and-read path, and auto-open a configured EPUB after boot for device debugging.

**Architecture:** Add a focused `ink_epub_reader` module that parses a narrow EPUB subset into UTF-8 pages, then reuse the current text-reader rendering, button navigation, and persisted page state. Keep the runtime shell unchanged wherever possible and confine new logic mostly to EPUB loading and app flow wiring.

**Tech Stack:** ESP-IDF 5.5.4, C, FATFS/SDMMC, existing `ink_txt_reader` page model, existing EPD render pipeline

Create `main/ink_text_page_builder.c` by extracting the current pagination and UTF-8 byte-consumption logic from `main/ink_txt_reader.c` with the same behavior:

```c
#include "ink_text_page_builder.h"

#include <string.h>

static void trim_trailing_spaces(char *text)
{
    size_t len = strlen(text);
    while (len > 0 && text[len - 1] == ' ') {
        text[--len] = '\0';
    }
}

static size_t utf8_expected_length(unsigned char lead)
{
    if (lead < 0x80U) {
        return 1;
    }
    if ((lead & 0xE0U) == 0xC0U) {
        return 2;
    }
    if ((lead & 0xF0U) == 0xE0U) {
        return 3;
    }
    if ((lead & 0xF8U) == 0xF0U) {
        return 4;
    }
    return 1;
}

static bool advance_position(ink_text_page_builder_t *builder)
{
    ink_txt_reader_t *reader = builder->reader;
    size_t len = strlen(reader->pages[builder->page_index][builder->line_index]);
    while (len > 0 && reader->pages[builder->page_index][builder->line_index][len - 1] == ' ') {
        reader->pages[builder->page_index][builder->line_index][--len] = '\0';
    }

    ++builder->line_index;
    builder->column = 0;
    if (builder->line_index < INK_TXT_READER_BODY_LINES) {
        return true;
    }

    ++builder->page_index;
    builder->line_index = 0;
    if (builder->page_index >= INK_TXT_READER_MAX_PAGES) {
        reader->truncated = true;
        builder->page_index = INK_TXT_READER_MAX_PAGES - 1;
        builder->line_index = INK_TXT_READER_BODY_LINES - 1;
        builder->column = strlen(reader->pages[builder->page_index][builder->line_index]);
        return false;
    }

    if (reader->page_count < builder->page_index + 1U) {
        reader->page_count = builder->page_index + 1U;
    }
    return true;
}

static bool append_bytes(ink_text_page_builder_t *builder, const uint8_t *bytes, size_t byte_count)
{
    ink_txt_reader_t *reader = builder->reader;
    if (builder->column + byte_count > INK_TXT_READER_LINE_LENGTH) {
        if (!advance_position(builder)) {
            return false;
        }
    }

    if (byte_count > INK_TXT_READER_LINE_LENGTH) {
        static const uint8_t fallback[] = "?";
        bytes = fallback;
        byte_count = 1;
    }

    char *line = reader->pages[builder->page_index][builder->line_index];
    memcpy(line + builder->column, bytes, byte_count);
    builder->column += byte_count;
    line[builder->column] = '\0';
    return true;
}

void ink_text_page_builder_begin(ink_text_page_builder_t *builder, ink_txt_reader_t *reader)
{
    memset(builder, 0, sizeof(*builder));
    builder->reader = reader;
}

bool ink_text_page_builder_append_utf8_byte(
    ink_text_page_builder_t *builder,
    ink_text_utf8_state_t *state,
    unsigned char raw)
{
    uint8_t bytes[4];

    if (builder == NULL || builder->reader == NULL || state == NULL) {
        return false;
    }

    if (raw == '\r') {
        return true;
    }

    if (raw == '\n') {
        state->expected = 0;
        state->length = 0;
        return advance_position(builder);
    }

    if (state->length == 0) {
        if (raw < 0x80U) {
            bytes[0] = raw == '\t' ? ' ' : raw;
            return append_bytes(builder, bytes, 1);
        }

        state->expected = utf8_expected_length(raw);
        state->bytes[0] = raw;
        state->length = 1;
        if (state->expected == 1) {
            state->length = 0;
            bytes[0] = '?';
            return append_bytes(builder, bytes, 1);
        }
        return true;
    }

    if ((raw & 0xC0U) == 0x80U && state->length < sizeof(state->bytes)) {
        state->bytes[state->length++] = raw;
        if (state->length == state->expected) {
            const bool ok = append_bytes(builder, state->bytes, state->length);
            state->expected = 0;
            state->length = 0;
            return ok;
        }
        return true;
    }

    bytes[0] = '?';
    state->expected = 0;
    state->length = 0;
    if (!append_bytes(builder, bytes, 1)) {
        return false;
    }
    if (raw < 0x80U) {
        bytes[0] = raw;
        return append_bytes(builder, bytes, 1);
    }
    return true;
}

bool ink_text_page_builder_append_utf8_bytes(
    ink_text_page_builder_t *builder,
    ink_text_utf8_state_t *state,
    const uint8_t *raw,
    size_t length,
    size_t *non_ascii_bytes)
{
    if (builder == NULL || builder->reader == NULL || state == NULL || raw == NULL) {
        return false;
    }

    for (size_t i = 0; i < length && !builder->reader->truncated; ++i) {
        if (raw[i] >= 0x80U && non_ascii_bytes != NULL) {
            ++(*non_ascii_bytes);
        }
        if (!ink_text_page_builder_append_utf8_byte(builder, state, raw[i])) {
            return false;
        }
    }

    return true;
}

void ink_text_page_builder_finish(
    ink_text_page_builder_t *builder,
    ink_text_utf8_state_t *state,
    bool exhausted)
{
    ink_txt_reader_t *reader = builder->reader;
    if (state != NULL && state->length > 0 && !reader->truncated) {
        static const uint8_t fallback[] = "?";
        (void)append_bytes(builder, fallback, 1);
    }

    for (size_t page = 0; page < reader->page_count; ++page) {
        for (size_t row = 0; row < INK_TXT_READER_BODY_LINES; ++row) {
            trim_trailing_spaces(reader->pages[page][row]);
        }
    }

    if (reader->pages[0][0][0] == '\0') {
        strcpy(reader->pages[0][0], exhausted ? "EMPTY TXT FILE" : "NO DISPLAY DATA");
    }
}

bool ink_text_page_builder_self_test(void)
{
    ink_txt_reader_t reader;
    ink_text_page_builder_t builder;
    ink_text_utf8_state_t state = {0};
    static const uint8_t sample[] =
        "A\n"
        "B\n"
        "C\n"
        "D\n"
        "E\n"
        "\xE4\xB8\xAD\n";

    ink_txt_reader_prepare_default(&reader);
    reader.loaded = true;
    strcpy(reader.title, "BUILDER");
    ink_text_page_builder_begin(&builder, &reader);

    if (!ink_text_page_builder_append_utf8_bytes(&builder, &state, sample, sizeof(sample) - 1U, &reader.non_ascii_bytes)) {
        return false;
    }

    ink_text_page_builder_finish(&builder, &state, true);

    if (strcmp(reader.pages[0][0], "A") != 0) {
        return false;
    }
    if (strcmp(reader.pages[1][1], "\xE4\xB8\xAD") != 0) {
        return false;
    }
    if (reader.page_count != 2U) {
        return false;
    }
    return true;
}
```
